function updateHome() { const now = new Date(); const currentMin = now.getHours() * 60 + now.getMinutes(); const currentSec = now.getSeconds(); const today = getToday(); const todayDay = today.getDate(); const todayMonth = today.getMonth(); const todayYear = today.getFullYear(); let todayIndex = scheduleData.findIndex(d => { const dayNum = parseInt(d.date.split(' ')[0]); return dayNum === todayDay && todayMonth === 1 && todayYear === 2026; }); if (todayIndex === -1) { todayIndex = 0; } const todayData = scheduleData[todayIndex]; if (!todayData) return; document.getElementById('home-imsak').textContent = to12Hour(todayData.imsak); document.getElementById('home-subuh').textContent = to12Hour(todayData.subuh); document.getElementById('home-maghrib').textContent = to12Hour(todayData.maghrib); document.getElementById('home-isya').textContent = to12Hour(todayData.isya); const imsakMin = timeToMinutes(todayData.imsak); const maghribMin = timeToMinutes(todayData.maghrib); const isyaMin = timeToMinutes(todayData.isya); let status, label, targetMin, progress; let showCountdown = true; let isNextDay = false; if (currentMin < imsakMin) { status = 'Sahur'; label = 'Menuju imsak'; targetMin = imsakMin; progress = 0; } else if (currentMin < maghribMin) { status = 'Berpuasa'; label = 'Menuju buka'; targetMin = maghribMin; progress = ((currentMin - imsakMin) / (maghribMin - imsakMin)) * 100; } else if (currentMin < isyaMin) { status = 'Buka Puasa'; label = 'Menuju isya'; targetMin = isyaMin; progress = 100; } else { status = 'Istirahat'; isNextDay = true; const nextDayIndex = todayIndex + 1; if (nextDayIndex < scheduleData.length) { const nextDayData = scheduleData[nextDayIndex]; label = 'Menuju imsak besok'; targetMin = timeToMinutes(nextDayData.imsak) + (24 * 60); progress = 100; } else { label = 'Selesai'; targetMin = isyaMin; progress = 100; showCountdown = false; } } document.getElementById('home-status').textContent = status; document.getElementById('home-countdown-label').textContent = label; const circle = document.getElementById('progress-circle'); const radius = circle.r.baseVal.value; const circumference = radius * 2 * Math.PI; const offset = circumference - (progress / 100) * circumference; circle.style.strokeDashoffset = offset; document.getElementById('progress-text').textContent = Math.round(progress) + '%'; if (showCountdown) { const nowTotalSec = currentMin * 60 + currentSec; let targetTotalSec; if (isNextDay && todayIndex + 1 < scheduleData.length) { const nextDayData = scheduleData[todayIndex + 1]; targetTotalSec = (24 * 60 * 60) + (timeToMinutes(nextDayData.imsak) * 60); } else { targetTotalSec = targetMin * 60; } let diffSec = targetTotalSec - nowTotalSec; if (diffSec < 0) diffSec = 0; const h = Math.floor(diffSec / 3600); const m = Math.floor((diffSec % 3600) / 60); const s = diffSec % 60; document.getElementById('home-countdown').textContent = `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`; } else { document.getElementById('home-countdown').textContent = '00:00:00'; } // PERBAIKAN: Logika untuk menentukan jadwal selanjutnya let nextPrayer = null; let isShowingTomorrow = false; let isShowingToday = false; // Cek semua jadwal sholat hari ini yang belum lewat const todayPrayers = [ { name: 'Imsak', time: todayData.imsak, done: currentMin > timeToMinutes(todayData.imsak) }, { name: 'Subuh', time: todayData.subuh, done: currentMin > timeToMinutes(todayData.subuh) }, { name: 'Dzuhur', time: todayData.dzuhur, done: currentMin > timeToMinutes(todayData.dzuhur) }, { name: 'Ashar', time: todayData.ashar, done: currentMin > timeToMinutes(todayData.ashar) }, { name: 'Maghrib', time: todayData.maghrib, done: currentMin > timeToMinutes(todayData.maghrib) }, { name: 'Isya', time: todayData.isya, done: currentMin > timeToMinutes(todayData.isya) } ].filter(p => !p.done); if (todayPrayers.length > 0) { // Masih ada jadwal hari ini yang belum lewat nextPrayer = todayPrayers[0]; isShowingToday = true; } else { // Semua jadwal hari ini sudah lewat, ambil Imsak besok isShowingTomorrow = true; const nextDayIndex = todayIndex + 1; if (nextDayIndex < scheduleData.length) { const nextDayData = scheduleData[nextDayIndex]; nextPrayer = { name: 'Imsak', time: nextDayData.imsak, done: false, isTomorrow: true }; } } // PERBAIKAN: Ambil nama hari yang benar dari data let html = ''; if (isShowingToday && nextPrayer) { // Ambil nama hari dari fullDate (format: "Minggu, 01 Februari 2026") const dayName = todayData.fullDate.split(',')[0]; html += `
📅 Sekarang (${dayName})
`; } else if (isShowingTomorrow && nextPrayer) { const nextDayData = scheduleData[todayIndex + 1]; const dayName = nextDayData.fullDate.split(',')[0]; html += `
📅 Besok (${dayName})
`; } if (nextPrayer) { const isMaghrib = nextPrayer.name === 'Maghrib'; const isTomorrow = nextPrayer.isTomorrow || false; const bgClass = isMaghrib ? 'bg-orange-500/20 border-2 border-orange-500/50' : 'bg-slate-800/50 border border-slate-700'; const nameClass = isMaghrib ? 'text-orange-400 font-bold text-base' : (isTomorrow ? 'text-blue-300 font-bold text-base' : 'text-white font-bold text-base'); const timeClass = isMaghrib ? 'text-orange-400 text-lg' : (isTomorrow ? 'text-blue-300 text-lg' : 'text-gray-300 text-lg'); const icon = isTomorrow ? '🌅 ' : '⏰ '; html += `
${icon}${nextPrayer.name} ${to12Hour(nextPrayer.time)}
`; } else { html = '

Tidak ada jadwal tersedia

'; } document.getElementById('next-prayer-list').innerHTML = html; }
